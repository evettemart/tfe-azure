  Certificate

  # Set your FQDN
FQDN="tfe.azure.wind.com"

# Generate a private key (no password — required by the README)
openssl genrsa -out privkey.pem 4096

# Generate a self-signed certificate valid for 1 year
openssl req -new -x509 \
  -key privkey.pem \
  -out cert.pem \
  -days 365 \
  -subj "/CN=${FQDN}" \
  -addext "subjectAltName=DNS:${FQDN}"

# The CA bundle for a self-signed cert is the cert itself
cp cert.pem ca_bundle.pem

cat cert.pem    | base64 > cert.b64
cat privkey.pem | base64 > privkey.b64
cat ca_bundle.pem | base64 > ca_bundle.b64

tr -d '\n' < cert.b64 > cert_nowrap.b64
tr -d '\n' < privkey.b64 > privkey_nowrap.b64
tr -d '\n' < ca_bundle.b64 > ca_bundle_nowrap.b64

az keyvault secret set \
  --vault-name "$KEYVAULT_NAME" \
  --name "tfe-tls-cert" \
  --file cert_nowrap.b64

  az keyvault secret set \
  --vault-name "$KEYVAULT_NAME" \
  --name "tfe-tls-privkey" \
  --file privkey_nowrap.b64

# CA bundle
az keyvault secret set \
  --vault-name "$KEYVAULT_NAME" \
  --name "tfe-tls-ca-bundle" \
  --file ca_bundle_nowrap.b64

Create a bastion host
1. Create subnet
az network vnet subnet create \
  --name "AzureBastionSubnet" \
  --resource-group "tfe-networking-rg" \
  --vnet-name "tfe-vnet" \
  --address-prefixes "10.0.5.0/26"

2. Creat public IP
az network public-ip create \
  --name "tfe-bastion-ip" \
  --resource-group "tfe-networking-rg" \
  --location "uksouth" \
  --sku Standard \
  --allocation-method Static \
  --zone 1 2 3

3. Create the bastion host
az network bastion create \
  --name "tfe-bastion" \
  --resource-group "tfe-networking-rg" \
  --location "uksouth" \
  --vnet-name "tfe-vnet" \
  --public-ip-address "tfe-bastion-ip" \
  --sku Basic

4. Wait for it to be ready - should say succeeded
az network bastion show \
  --name "tfe-bastion" \
  --resource-group "tfe-networking-rg" \
  --query "provisioningState" -o tsv

5. Delete when done
az network bastion delete \
  --name "tfe-bastion" \
  --resource-group "tfe-networking-rg"
